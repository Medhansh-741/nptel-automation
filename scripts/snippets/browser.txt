# ----------------------------------------------------------------------------
# 0. VM-ESCAPE HEADER (REQUIRED in every run_code_unsafe call that touches fs)
# ----------------------------------------------------------------------------
# The run_code_unsafe sandbox strips Node globals (require/process/Buffer/import).
# Escape via the Function constructor to reach the real Node process:
#   const nodeProcess = globalThis.constructor.constructor('return process')()
#                     .mainModule.require;
#   const fs   = nodeProcess('fs');
#   const path = nodeProcess('path');
#   const Buffer = nodeProcess('buffer').Buffer;
#
# Without this you WILL hit: "ReferenceError: require is not defined" /
# "process is not defined" / "Buffer is not defined". This is expected; use the escape.
